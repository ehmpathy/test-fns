# blackbox criteria: progress reporter

## usecase.1 = monitor test progress in real-time

```
given('a jest test suite configured with progress reporter')
  when('tests are mid-run')
    then('progress.overview.jsonl contains heartbeat lines every ~1s')
      sothat('external tools can poll for progress')
    then('each heartbeat line has passed, failed, skipped, remains, outof counts')
    then('each heartbeat line has duration in ISO 8601 format (PT2M34S)')
    then('remains + passed + failed + skipped = outof')
      sothat('counts are internally consistent')

given('tests complete')
  when('final state emitted')
    then('last heartbeat line shows remains = 0')
    then('passed + failed + skipped = outof')
    then('progress.summary.json written with final state')
      sothat('tools can read final state without parse of jsonl')
```

## usecase.2 = react to failures immediately

```
given('a test fails')
  when('failure detected by reporter')
    then('failure appended to progress.failures.jsonl immediately')
      sothat('external tools can react before suite completes')
    then('failure line has path (file path)')
    then('failure line has suite (full test title with hierarchy)')
    then('failure line has failure (assertion errors + stack traces from runner)')
    then('failure line has stdout (console.log output from run)')
    then('failure line has stderr (console.warn/error output from run)')

given('test with large console output fails')
  when('failure appended')
    then('stdout truncated to 10kb max')
    then('stderr truncated to 10kb max')
    then('failure field preserved in full')
      sothat('failures.jsonl doesnt bloat unboundedly')

given('no tests fail')
  when('tests complete')
    then('progress.failures.jsonl exists but is empty')
      sothat('tools can watch the file without special-case for absence')
```

## usecase.3 = configure reporter via jest.config

```
given('jest.config.js with progress reporter and dir option')
  when('tests run')
    then('reporter creates dir if absent')
    then('reporter writes to dir/progress.overview.jsonl')
    then('reporter writes to dir/progress.failures.jsonl')
    then('reporter writes to dir/progress.summary.json on completion')

given('progress reporter without dir option')
  when('tests run')
    then('reporter throws clear error about required dir config')
      sothat('misconfiguration fails fast with actionable message')

given('dir path with nested structure that doesnt exist')
  when('tests run')
    then('full directory tree created')
    then('files written successfully')
```

## usecase.4 = support append for resume

```
given('prior progress files exist from previous run')
  when('new run starts')
    then('new heartbeat lines appended to progress.overview.jsonl')
    then('new failures appended to progress.failures.jsonl')
      sothat('crash recovery shows history + current state')
    then('progress.summary.json overwritten with current run state')
      sothat('summary always reflects the latest complete run')

given('progress.overview.jsonl grows across multiple runs')
  when('external tool reads file')
    then('each line is valid json')
    then('lines are newline-delimited')
    then('tool can tail -1 to get latest state')
```

## usecase.5 = track skipped tests

```
given('tests with skipped or todo tests')
  when('heartbeat emitted')
    then('skipped count includes skipped and todo tests')
    then('skipped tests do not appear in progress.failures.jsonl')
    then('skipped tests do not affect passed or failed counts')

given('all tests are skipped')
  when('tests complete')
    then('passed = 0, failed = 0, skipped = outof, remains = 0')
    then('progress.failures.jsonl empty')
```

## usecase.6 = handle edge cases

```
given('test suite crashes mid-run')
  when('process terminates')
    then('last heartbeat line shows state before crash')
    then('failures up to crash point are in failures.jsonl')
      sothat('partial progress is preserved for diagnosis')

given('test with no assertions')
  when('tests complete')
    then('test counted as passed')
    then('no entry in failures.jsonl')

given('test file that throws in module scope')
  when('tests complete')
    then('failure captured in failures.jsonl')
    then('suite field shows file path')
    then('reporter does not crash')

given('1000+ test files')
  when('tests complete')
    then('all heartbeats written')
    then('all failures captured')
    then('performance remains acceptable')
      sothat('progress reporter doesnt bottleneck test runs')

given('tests run in parallel workers')
  when('results aggregated')
    then('passed/failed counts reflect all workers')
    then('failures from all workers captured')
    then('no race conditions on file writes')
      sothat('parallel execution works correctly')
```

## usecase.7 = provide final summary

```
given('tests complete successfully')
  when('progress.summary.json read')
    then('json has passed count')
    then('json has failed count')
    then('json has skipped count')
    then('json has outof count')
    then('json has duration in ISO 8601 format')
      sothat('final state is easy to parse programmatically')

given('tests complete with failures')
  when('progress.summary.json read')
    then('failed count matches failure entries in failures.jsonl')
    then('passed + failed + skipped = outof')
```

## usecase.8 = console output separation

```
given('a failed test with console.log statements')
  when('failure captured')
    then('console.log output appears in stdout field')
    then('console.log output does not appear in stderr field')

given('a failed test with console.warn statements')
  when('failure captured')
    then('console.warn output appears in stderr field')
    then('console.warn output does not appear in stdout field')

given('a failed test with console.error statements')
  when('failure captured')
    then('console.error output appears in stderr field')
    then('console.error output does not appear in stdout field')

given('a failed test with mixed console output')
  when('failure captured')
    then('console.log → stdout')
    then('console.warn → stderr')
    then('console.error → stderr')
    then('assertion errors and stack traces → failure')
```
